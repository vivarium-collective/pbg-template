**Goal:** Replace the loose "Phase" concept in the Build Model tab with a structured "Task" — a discrete, measurable model-development experiment with schema, migration, CRUD endpoints, UI, and a run-baseline endpoint.

**Architecture:** The schema grows a new `task` definition alongside (not replacing) the legacy `phase` definition, keeping backward compatibility. Server endpoints follow the existing pattern (`_active_branch_action` for mutations, `do_POST`/`do_DELETE` dispatch tables). The UI replaces the phase tracker cards in the Build Model page with task cards rendered by `report.py` → `index.html.j2`.

**Tech Stack:** Python 3.11+, PyYAML, jsonschema, process-bigraph, Jinja2, vanilla JS (no frameworks), CSS custom classes.

Create `/Users/eranagmon/code/pbg-template/scripts/_migrate_phases_to_tasks.py` with this exact content:

```python
"""One-shot migration: workspace.yaml.phases → workspace.yaml.tasks.

For each phase entry, build a corresponding task with:
- n, name, status copied across (status: 'planned' stays 'planned'; 'in_progress' → 'baseline_run'; 'complete' → 'passed')
- objective: extracted from phase-<n>.md frontmatter or body if available
- acceptance_criteria: copied from phase frontmatter if present (legacy phases had a list)
- simulation, datasets, etc.: empty/None (user fills in)

Writes the new tasks list to workspace.yaml; leaves phases as-is for backward
compat (user can delete phases manually once they're satisfied with tasks).

Run from workspace root:
    python3 scripts/_migrate_phases_to_tasks.py
"""
from __future__ import annotations
import sys
from pathlib import Path

import yaml


STATUS_MAP = {
    "planned": "planned",
    "in_progress": "baseline_run",  # was actively being worked on
    "gate_pending": "perturbed_run",  # was awaiting evaluation
    "complete": "passed",
}


def migrate(ws_root: Path) -> int:
    ws_file = ws_root / "workspace.yaml"
    ws = yaml.safe_load(ws_file.read_text())
    phases = ws.get("phases") or []
    if not phases:
        print("no phases to migrate")
        return 0

    existing_tasks = {t["n"]: t for t in (ws.get("tasks") or [])}
    new_tasks = []
    for p in phases:
        n = p.get("n")
        if n is None:
            continue
        if n in existing_tasks:
            print(f"task {n} already exists; skipping migration of phase {n}")
            continue
        task = {
            "n": n,
            "name": p.get("name", f"phase-{n}").lower().replace(" ", "-"),
            "status": STATUS_MAP.get(p.get("status", "planned"), "planned"),
        }
        # Try to pick up phase-<n>.md objective if present
        phase_md = ws_root / "phases" / f"phase-{n}.md"
        if phase_md.exists():
            text = phase_md.read_text()
            # Crude extract: line after "### Objective"
            try:
                obj_idx = text.index("### Objective")
                rest = text[obj_idx:].splitlines()[1:6]
                objective = " ".join(line.strip() for line in rest if line.strip() and not line.startswith("#"))
                if objective:
                    task["objective"] = objective[:300]
            except ValueError:
                pass
        if p.get("acceptance_tests"):
            task["acceptance_criteria"] = [
                {"perturbation": str(a.get("desc", a.get("id", ""))),
                 "expected": "(filled in by user)"}
                for a in p["acceptance_tests"]
            ]
        new_tasks.append(task)

    ws["tasks"] = (ws.get("tasks") or []) + new_tasks
    ws_file.write_text(yaml.safe_dump(ws, sort_keys=False))
    print(f"migrated {len(new_tasks)} phase(s) to task(s)")
    print("Note: workspace.yaml.phases is left intact for backward compat. Delete it manually when ready.")
    return 0


if __name__ == "__main__":
    sys.exit(migrate(Path.cwd()))
```

    def _post_task(self, body: dict):

        Body: {name, objective?, simulation?, datasets?, expert_docs?,
               visualizations?, model_changes?, desired_behaviors?, acceptance_criteria?}

    def _patch_task(self, body: dict):
        """Partial update of a task.

        Body: {n, ...fields...}  — only provided fields are updated.
        Status transitions are validated (no skipping multiple levels).
        """
        n = body.get("n")
        if not isinstance(n, int):
            return self._json({"error": "n must be an integer"}, 400)

        new_status = body.get("status")
        if new_status is not None and new_status not in self._VALID_TASK_STATUSES:
            return self._json({"error": f"invalid status '{new_status}'; valid: {sorted(self._VALID_TASK_STATUSES)}"}, 400)

        ref_errors = self._validate_task_refs(body)
        if ref_errors:
            return self._json({"error": "invalid cross-references", "details": ref_errors}, 400)

        def action():
            _ws_add_to_sys_path()
            from scripts._lib.workspace_yaml import load_workspace, save_workspace
            ws_file = WORKSPACE / "workspace.yaml"
            ws = load_workspace(ws_file)
            task = next((t for t in (ws.get("tasks") or []) if t.get("n") == n), None)
            if task is None:
                raise ValueError(f"task n={n} not found")
            # Status transition check: cannot skip more than 2 steps forward
            if new_status is not None:
                try:
                    old_idx = self._TASK_STATUS_ORDER.index(task.get("status", "planned"))
                    new_idx = self._TASK_STATUS_ORDER.index(new_status)
                    if new_idx > old_idx + 2:
                        raise ValueError(
                            f"invalid status transition: '{task.get('status')}' → '{new_status}' "
                            f"(skips too many steps)"
                        )
                except ValueError as ve:
                    if "invalid status transition" in str(ve):
                        raise
                task["status"] = new_status
            for field in ("objective", "simulation", "baseline_run", "perturbed_run", "comparison"):
                if field in body:
                    val = body[field]
                    if val is None or val == "":
                        task.pop(field, None)
                    else:
                        task[field] = val
            for field in ("datasets", "expert_docs", "visualizations", "model_changes",
                          "desired_behaviors", "acceptance_criteria"):
                if field in body:
                    val = body[field]
                    if val is None or (isinstance(val, list) and not val):
                        task.pop(field, None)
                    else:
                        task[field] = val
            save_workspace(ws_file, ws)

        return self._json(*_active_branch_action(f"feat(task-{n}): update task", action))
